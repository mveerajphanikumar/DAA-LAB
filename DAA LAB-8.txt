#include <iostream>
#include <vector>
#include <queue>
#include <chrono>
using namespace std;
using namespace chrono;

vector<vector<int>> graph;
vector<bool> visited;

void DFS(int v) {
    visited[v] = true;
    cout << v << " ";

    for (int n : graph[v])
        if (!visited[n])
            DFS(n);
}

void BFS(int start) {
    queue<int> q;
    visited.assign(graph.size(), false);
	

    visited[start] = true;
    q.push(start);

    while (!q.empty()) {
        int v = q.front();
        q.pop();

        cout << v << " ";

        for (int n : graph[v]) {
            if (!visited[n]) {
                visited[n] = true;
                q.push(n);
            }
        }
    }
}

int main() {
    int V, E;

    cout << "Enter number of vertices: ";
    cin >> V;

    graph.resize(V);

    cout << "Enter number of edges: ";
    cin >> E;

    cout << "Enter edges (u v):\n";

    for (int i = 0; i < E; i++) {
        int u, v;
        cin >> u >> v;

        graph[u].push_back(v);
        graph[v].push_back(u);
    }

    int start;
    cout << "Enter starting vertex: ";
    cin >> start;

    // DFS
    visited.assign(V, false);

    auto startDFS = high_resolution_clock::now();

    cout << "\nDFS Traversal: ";
    DFS(start);

    auto endDFS = high_resolution_clock::now();

    // BFS
    auto startBFS = high_resolution_clock::now();

    cout << "\n\nBFS Traversal: ";
    BFS(start);

    auto endBFS = high_resolution_clock::now();

    auto dfsTime = duration_cast<nanoseconds>(endDFS - startDFS);
    auto bfsTime = duration_cast<nanoseconds>(endBFS - startBFS);

    cout << "\n\nExecution Time:";
    cout << "\nDFS: " << dfsTime.count() << " ns";
    cout << "\nBFS: " << bfsTime.count() << " ns";

    return 0;
}
